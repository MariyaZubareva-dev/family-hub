function addMonths(base:string,months:number,day:number){const d=new Date(`${base}T00:00:00`);const target=new Date(d.getFullYear(),d.getMonth()+months,1);const last=new Date(target.getFullYear(),target.getMonth()+1,0).getDate();return `${target.getFullYear()}-${String(target.getMonth()+1).padStart(2,'0')}-${String(Math.min(day,last)).padStart(2,'0')}`;}

function annuity(principal:number,annualRate:number,months:number){const r=annualRate/100/12;return r>0&&months>0?principal*r/(1-Math.pow(1+r,-months)):months>0?principal/months:0;}
function monthsForPayment(balance:number,annualRate:number,payment:number){if(balance<=0)return 0;const r=annualRate/100/12;if(payment<=0)return Infinity;if(r===0)return Math.ceil(balance/payment);if(payment<=balance*r)return Infinity;return Math.ceil(-Math.log(1-balance*r/payment)/Math.log(1+r));}
function payoffDate(start:string,paymentDay:number,balance:number,annualRate:number,payment:number){const months=monthsForPayment(balance,annualRate,payment);return Number.isFinite(months)?addMonths(start,months,paymentDay):undefined;}
function payoffDateFromPaymentDate(paymentDate:string,balance:number,annualRate:number,payment:number){const months=monthsForPayment(balance,annualRate,payment);return Number.isFinite(months)?addMonthsFromDate(paymentDate,months):undefined;}
function isLeapYear(year:number){return year%4===0&&(year%100!==0||year%400===0);}
function daysInYearValue(year:number){return isLeapYear(year)?366:365;}
function dailyAccruedInterest(balance:number,annualRate:number,fromDate:string,toDate:string){if(toDate<=fromDate)return 0;let cursor=new Date(`${fromDate}T00:00:00`);const end=new Date(`${toDate}T00:00:00`);let interest=0;while(cursor<end){const nextDay=new Date(cursor.getTime()+86400000);const dayEnd=nextDay<end?nextDay:end;const days=Math.max(0,(dayEnd.getTime()-cursor.getTime())/86400000);interest+=balance*(annualRate/100)*days/daysInYearValue(cursor.getFullYear());cursor=nextDay;}return interest;}

function normalizeDateOnly(value:string):string{
  const s=String(value ?? '').trim();
  if(!s) return '';
  const m=s.match(/^(\d{4}-\d{2}-\d{2})/);
  return m ? m[1] : '';
}

function dateRu(value:string):string{
  const dateOnly=normalizeDateOnly(value);
  if(!dateOnly) return '—';
  const d=new Date(`${dateOnly}T00:00:00`);
  if(Number.isNaN(d.getTime())) return '—';
  return d.toLocaleDateString('ru-RU',{day:'numeric',month:'short',year:'numeric'});
}

function money(v:number):string{
  return v.toLocaleString('ru-RU',{minimumFractionDigits:2,maximumFractionDigits:2});
}

function Metric(p:{title:string;value:number;unit?:string}){
  return <div className="metric"><span>{p.title}</span><strong>{money(p.value)}{p.unit ? ` ${p.unit}` : ' ₽'}</strong></div>;
}

function useCreditSchedule(credit:Credit,mode:'TERM'|'PAYMENT'){
 return useMemo(()=>{
  const principal=num(credit.principal);
  const annualRate=num(credit.annual_rate);
  const basePayment=num(credit.standard_payment);
  const originalTerm=Math.max(1,Math.round(num(credit.term_months)));
  const asOfDate=todayString();
  const firstPaymentDate=normalizeDateOnly(credit.first_payment_date);
  const startDate=normalizeDateOnly(credit.start_date);
  const monthlyRate=annualRate/100/12;
  const historicalPrepayments=[...credit.prepayments]
    .map(pp=>({id:pp.id,amount:num(pp.amount),paid_on:normalizeDateOnly(pp.paid_on),comment:pp.comment}))
    .filter(pp=>pp.amount>0&&pp.paid_on>=startDate&&pp.paid_on<=asOfDate)
    .sort((a,b)=>a.paid_on.localeCompare(b.paid_on));

  let balance=principal;
  let payment=basePayment;
  let totalInterest=0;
  let month=0;
  let paidMonths=0;
  let currentBalance=principal;
  let accruedInterestAsOf=0;
  let nextPaymentDate:string|undefined;
  let nextPaymentAmount=0;
  let currentCaptured=false;
  let currentInterestAnchor=startDate;
  let currentPeriodHadMidPrepayment=false;
  let periodInterestAnchor=firstPaymentDate;
  let useDailyInterestForPeriod=false;
  const rows:ScheduleRow[]=[];
  const originalEndDate=addMonthsFromDate(firstPaymentDate,originalTerm-1);
  const contractualPayment=annuity(principal,annualRate,originalTerm);

  while(balance>0.01&&month<originalTerm+1200){
    month++;
    const dueDate=addMonthsFromDate(firstPaymentDate,month-1);
    const nextDueDate=addMonthsFromDate(firstPaymentDate,month);
    const isHistorical=dueDate<=asOfDate;

    if(!isHistorical&&!currentCaptured){
      currentBalance=balance;
      nextPaymentDate=dueDate;
      nextPaymentAmount=Math.min(payment,balance + (useDailyInterestForPeriod
        ? dailyAccruedInterest(balance,annualRate,periodInterestAnchor,dueDate)
        : balance*monthlyRate));
      accruedInterestAsOf=asOfDate>periodInterestAnchor
        ? dailyAccruedInterest(balance,annualRate,periodInterestAnchor,asOfDate)
        : 0;
      currentCaptured=true;
    }

    const regularInterest=useDailyInterestForPeriod
      ? dailyAccruedInterest(balance,annualRate,periodInterestAnchor,dueDate)
      : balance*monthlyRate;
    const regularPayment=Math.min(payment,balance+regularInterest);
    const regularPrincipal=Math.min(balance,Math.max(0,regularPayment-regularInterest));
    balance-=regularPrincipal;
    totalInterest+=regularInterest;

    let prepayment=0;
    let prepaymentInterest=0;
    let prepaymentPrincipal=0;
    const prepaymentDates:string[]=[];
    let lastPrepaymentDate:string|undefined;

    if(isHistorical){
      const periodPrepayments=historicalPrepayments.filter(pp=>pp.paid_on>=dueDate&&pp.paid_on<nextDueDate);
      let segmentStart=dueDate;

      for(const pp of periodPrepayments){
        if(balance<=0.01) break;
        const sameDay=pp.paid_on===dueDate;
        const accrued=sameDay?0:dailyAccruedInterest(balance,annualRate,segmentStart,pp.paid_on);
        const interestPart=Math.min(pp.amount,Math.max(0,accrued));
        const principalPart=Math.min(balance,Math.max(0,pp.amount-interestPart));

        prepayment+=pp.amount;
        prepaymentInterest+=interestPart;
        prepaymentPrincipal+=principalPart;
        prepaymentDates.push(pp.paid_on);
        totalInterest+=interestPart;
        balance-=principalPart;
        lastPrepaymentDate=pp.paid_on;
        segmentStart=pp.paid_on;
      }
    }

    balance=Math.max(0,balance);
    const totalPaid=regularPayment+prepayment;
    let newPayment:number|undefined;
    let paymentChangeDate:string|undefined;
    let newEndDate:string|undefined;

    if(prepayment>0.009&&balance>0.01&&isHistorical){
      if(mode==='PAYMENT'){
        const remainingPayments=Math.max(1,originalTerm-month);
        newPayment=annuity(balance,annualRate,remainingPayments);
        payment=newPayment;
        paymentChangeDate=nextDueDate;
      }else{
        const n=monthsForPayment(balance,annualRate,payment);
        if(Number.isFinite(n)&&n>0)newEndDate=addMonthsFromDate(nextDueDate,Math.max(0,n-1));
      }
    }

    rows.push({month,dueDate,payment:regularPayment,interest:regularInterest,principal:regularPrincipal,prepayment,prepaymentInterest,prepaymentPrincipal,totalPaid,balance,newPayment,paymentChangeDate,newEndDate,prepaymentDates:prepaymentDates.length?prepaymentDates:undefined});

    // The next period normally uses the monthly annuity interest. If a prepayment
    // happened after the due date, the next regular payment is based on the actual
    // accrued days from that prepayment date to the next due date.
    if(lastPrepaymentDate&&lastPrepaymentDate!==dueDate){
      periodInterestAnchor=lastPrepaymentDate;
      useDailyInterestForPeriod=true;
    }else{
      periodInterestAnchor=nextDueDate;
      useDailyInterestForPeriod=false;
    }

    if(isHistorical){
      paidMonths=Math.max(paidMonths,month);
      currentInterestAnchor=periodInterestAnchor;
      currentPeriodHadMidPrepayment=useDailyInterestForPeriod;
    }

    if(isHistorical&&dueDate===asOfDate){
      currentBalance=balance;
      currentCaptured=true;
      nextPaymentDate=nextDueDate;
      nextPaymentAmount=Math.min(payment,balance + (useDailyInterestForPeriod
        ? dailyAccruedInterest(balance,annualRate,periodInterestAnchor,nextDueDate)
        : balance*monthlyRate));
      accruedInterestAsOf=0;
    }

    if(balance<=0.01){
      if(isHistorical) currentBalance=0;
      break;
    }
  }

  if(!currentCaptured){
    if(balance<=0.01){
      currentBalance=0;
      nextPaymentDate=undefined;
      nextPaymentAmount=0;
      accruedInterestAsOf=0;
    }else{
      currentBalance=balance;
      nextPaymentDate=addMonthsFromDate(firstPaymentDate,month);
      const pendingInterest=useDailyInterestForPeriod
        ? dailyAccruedInterest(balance,annualRate,periodInterestAnchor,nextPaymentDate)
        : balance*monthlyRate;
      nextPaymentAmount=Math.min(payment,balance+pendingInterest);
      accruedInterestAsOf=asOfDate>periodInterestAnchor
        ? dailyAccruedInterest(balance,annualRate,periodInterestAnchor,asOfDate)
        : 0;
    }
  }

  let projectedEndDate:string|undefined;
  if(balance<=0.01&&rows.length) projectedEndDate=rows[rows.length-1].dueDate;
  else if(mode==='PAYMENT') projectedEndDate=originalEndDate;
  else if(nextPaymentDate&&currentBalance>0.01){
    const n=monthsForPayment(currentBalance,annualRate,payment);
    if(Number.isFinite(n)&&n>0) projectedEndDate=addMonthsFromDate(nextPaymentDate,Math.max(0,n-1));
  }

  return {rows,currentBalance,totalInterest,endDate:projectedEndDate??originalEndDate,originalEndDate,projectedEndDate,accruedInterestAsOf,nextPaymentDate,nextPaymentAmount,asOfDate,paidMonths,contractualPayment};
